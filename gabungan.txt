package org.delcom.ifs25028pbo2026springbootcashflow.adapter.presenter;

import org.delcom.ifs25028pbo2026springbootcashflow.domain.entity.Transaction;
import org.delcom.ifs25028pbo2026springbootcashflow.domain.entity.TransactionType;

import java.time.format.DateTimeFormatter;
import java.util.List;
import java.util.Locale;
import java.util.stream.Collectors;

public class FinancePresenter {

    private static final DateTimeFormatter DATE_TIME_FORMAT =
            DateTimeFormatter.ofPattern("dd/MM/yyyy HH:mm");

    public String formatMoney(double amount) {
        return "Rp " + String.format(Locale.US, "%,.2f", amount);
    }

    public String formatType(TransactionType type) {
        return type == TransactionType.INCOME ? "Pemasukan" : "Pengeluaran";
    }

    public String formatCreatedAt(Transaction transaction) {
        return transaction.getCreatedAt().format(DATE_TIME_FORMAT);
    }

    public String format(Transaction transaction) {
        return String.format("[%d] %s | %s | [%s]",
                transaction.getId(),
                transaction.getDescription(),
                formatMoney(transaction.getAmount()),
                formatType(transaction.getType()));
    }

    public String formatBalance(double balance) {
        return "Saldo: " + formatMoney(balance);
    }

    public String formatList(List<Transaction> transactions, String header, String emptyMessage) {
        StringBuilder builder = new StringBuilder(header);
        if (transactions.isEmpty()) {
            builder.append('\n').append(emptyMessage);
            return builder.toString();
        }
        String body = transactions.stream()
                .map(this::format)
                .collect(Collectors.joining("\n"));
        builder.append('\n').append(body);
        return builder.toString();
    }

    public String showTransactions(List<Transaction> transactions) {
        return formatList(transactions, "Daftar Transaksi:", "- Belum ada transaksi!");
    }

    public String showSearchResults(List<Transaction> transactions, String keyword) {
        return formatList(transactions, "Hasil Pencarian: \"" + keyword + "\"",
                "- Transaksi tidak ditemukan!");
    }

    public String showAddSuccess(Transaction transaction) {
        return "Berhasil menambah transaksi: " + format(transaction);
    }

    public String showRemoveSuccess() {
        return "Berhasil menghapus transaksi.";
    }

    public String showRemoveFailed(int id) {
        return String.format("[!] Gagal menghapus transaksi dengan ID: %d.", id);
    }

    public String showInvalidAmount() {
        return "[!] Jumlah harus berupa angka!";
    }

    public String showValidationError(String message) {
        return "[!] " + message;
    }
}

package org.delcom.ifs25028pbo2026springbootcashflow.adapter.repository;

import org.delcom.ifs25028pbo2026springbootcashflow.domain.repository.IRepository;

import java.util.ArrayList;
import java.util.List;
import java.util.Optional;
import java.util.function.Function;
import java.util.function.Predicate;

/**
 * Basis generic repository in-memory.
 *
 * @param <T>  tipe entitas
 * @param <ID> tipe identifier
 */
public abstract class InMemoryRepository<T, ID> implements IRepository<T, ID> {

    protected final List<T> data = new ArrayList<>();

    private final Function<T, ID> idExtractor;

    protected InMemoryRepository(Function<T, ID> idExtractor) {
        this.idExtractor = idExtractor;
    }

    @Override
    public List<T> findAll() {
        return new ArrayList<>(data);
    }

    @Override
    public Optional<T> findById(ID id) {
        return data.stream()
                .filter(item -> idExtractor.apply(item).equals(id))
                .findFirst();
    }

    @Override
    public T save(T entity) {
        data.add(entity);
        return entity;
    }

    @Override
    public boolean deleteById(ID id) {
        return data.removeIf(item -> idExtractor.apply(item).equals(id));
    }

    @Override
    public void update(T entity) {
    }

    @Override
    public List<T> findBy(Predicate<T> predicate) {
        return data.stream()
                .filter(predicate)
                .toList();
    }
}

    INCOME_FIRST(Comparator.comparing(Transaction::getType)),

    /** Pengeluaran ditampilkan lebih dulu. */
    EXPENSE_FIRST(Comparator.comparing(Transaction::getType).reversed());

    private final Comparator<Transaction> comparator;

    SortOption(Comparator<Transaction> comparator) {
        this.comparator = comparator;
    }

    public Comparator<Transaction> comparator() {
        return comparator;
    }

    public Transaction(int id, String description, double amount, TransactionType type,
                       LocalDateTime createdAt) {
        this.id = id;
        this.description = description;
        this.amount = amount;
        this.type = type;
        this.createdAt = createdAt;
    }

    public int getId() {
        return id;
    }

    public String getDescription() {
        return description;
    }

    public double getAmount() {
        return amount;
    }

    public TransactionType getType() {
        return type;
    }

    public LocalDateTime getCreatedAt() {
        return createdAt;
    }

    /** Mengubah keterangan transaksi. */
    public void changeDescription(String description) {
        this.description = description;
    }

    /** Mengubah jumlah uang transaksi. */
    public void changeAmount(double amount) {
        this.amount = amount;
    }

    /** Jumlah bertanda: positif untuk pemasukan, negatif untuk pengeluaran. */

package org.delcom.ifs25028pbo2026springbootcashflow.usecase;

import org.delcom.ifs25028pbo2026springbootcashflow.domain.entity.SortOption;
import org.delcom.ifs25028pbo2026springbootcashflow.domain.entity.Transaction;
import org.delcom.ifs25028pbo2026springbootcashflow.domain.entity.TransactionType;
import org.delcom.ifs25028pbo2026springbootcashflow.domain.exception.EntityNotFoundException;
import org.delcom.ifs25028pbo2026springbootcashflow.domain.exception.ValidationException;
import org.delcom.ifs25028pbo2026springbootcashflow.domain.repository.ITransactionRepository;

import java.util.List;

public class FinanceUseCase {

    private final ITransactionRepository repository;

    public FinanceUseCase(ITransactionRepository repository) {
        this.repository = repository;
    }

    public List<Transaction> getAllTransactions() {
        return repository.findAll();
    }

    public Transaction addIncome(String description, double amount) throws ValidationException {
        return add(description, amount, TransactionType.INCOME);
    }

    public Transaction addExpense(String description, double amount) throws ValidationException {
        return add(description, amount, TransactionType.EXPENSE);
    }

    private Transaction add(String description, double amount, TransactionType type)
            throws ValidationException {
        validate(description, amount);
        return repository.save(description.trim(), amount, type);
    }

    private void validate(String description, double amount) throws ValidationException {
        if (description == null || description.trim().isEmpty()) {
            throw new ValidationException("Keterangan tidak boleh kosong");
        }
        if (Double.isNaN(amount) || Double.isInfinite(amount) || amount <= 0) {
            throw new ValidationException("Jumlah harus lebih besar dari 0");
        }
    }

    public void removeTransaction(int id) throws EntityNotFoundException, ValidationException {
        if (id <= 0) {
            throw new ValidationException("ID tidak valid");
        }
        if (!repository.deleteById(id)) {
            throw new EntityNotFoundException(id);
        }
    }

    public List<Transaction> searchTransactions(String keyword) {
        String lower = keyword.toLowerCase();
        return repository.findBy(t -> t.getDescription().toLowerCase().contains(lower));
    }

    public List<Transaction> sortTransactions(SortOption option) {
        return repository.findAll().stream()
                .sorted(option.comparator())
                .toList();
    }

    public double getTotalIncome() {
        return totalOf(TransactionType.INCOME);
    }

    public double getTotalExpense() {
        return totalOf(TransactionType.EXPENSE);
    }

    public double getBalance() {
        return getTotalIncome() - getTotalExpense();
    }

    private double totalOf(TransactionType type) {
        return repository.findAll().stream()
                .filter(t -> t.getType() == type)
                .mapToDouble(Transaction::getAmount)
                .sum();
    }
}

import java.util.List;

/**
 * Membaca file {@code .env} di working directory dan menaruh isinya ke system properties
 * agar bisa dipakai di {@code application.yaml} (mis. {@code ${APP_PORT:8080}}).
 */
final class DotEnv {

    private DotEnv() {
    }

    static void load() throws IOException {
        load(Path.of(".env"));
    }

    static void load(Path path) throws IOException {
        if (!Files.isRegularFile(path)) {
            return;
        }
        List<String> lines = Files.readAllLines(path);
        for (String raw : lines) {
            applyLine(raw);
        }
    }

    static void applyLine(String raw) {
        String line = raw.trim();
        if (line.isEmpty() || line.startsWith("#")) {
            return;
        }
        int eq = line.indexOf('=');
        if (eq <= 0) {
            return;
        }
        String key = line.substring(0, eq).trim();
        String value = stripQuotes(line.substring(eq + 1).trim());
        if (System.getenv(key) != null || System.getProperty(key) != null) {
            return;
        }
        System.setProperty(key, value);
    }

    private static String stripQuotes(String value) {
        if (value.length() >= 2) {
            char first = value.charAt(0);
            char last = value.charAt(value.length() - 1);
            if ((first == '"' && last == '"') || (first == '\'' && last == '\'')) {
                return value.substring(1, value.length() - 1);
            }
        }
        return value;
    }
}
